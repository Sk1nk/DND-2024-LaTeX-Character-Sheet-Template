% New fitbox, based on the replies here:
%% https://tex.stackexchange.com/questions/754560/textbox-with-two-columns-that-adjusts-font-size-but-also-fill-one-column-before/754579#754579
% microtype package was already loaded elsewhere so I just needed to add the options

\microtypesetup{final,expansion=false,protrusion=true}

\setlength{\columnsep}{12pt} % gap between columns

% ---------- internals (no '@') ----------
\newdimen\fitFontDim
\newdimen\fitUpper
\newdimen\fitLower
\newdimen\fitMeasureDim
\newif\iffitMore
\newbox\fitTmp
\newbox\fitColA
\newbox\fitColB
\newdimen\fitLineWidth
\newbox\fitSingle
% ----------------------------------------

% \begin{fitbox}{<width>}{<height>}{<font switch>} ... \end{fitbox}
\NewDocumentEnvironment{fitbox2}{mmm +b}
{%
  % local line-breaking policy: allow hyphenation and enable microtype
  \microtypesetup{activate=true}%
  \vbadness=10000
  \hbadness=10000
  \tolerance=2000           % moderate
  \pretolerance=100
  \emergencystretch=.6em    % small emergency stretch (visually unobtrusive)
  \hyphenpenalty=50
  \exhyphenpenalty=50
  %
  #3 \fitFontDim\the\fontdimen6\font\relax
  \def\fitStuff{#4}%
  % --- build step: two columns without interline stretch ---
  \def\fitBuild{%
    \setbox\fitTmp=\vbox{%
      \hsize=\dimexpr(#1-\columnsep)/2\relax
      \parskip=0pt
      \fontsize{\fitFontDim}{1.2\fitFontDim}\selectfont
      \fitStuff\par
    }%
    \splittopskip=0pt \splitmaxdepth=0pt
    \setbox\fitColA=\vsplit\fitTmp to #2
    \setbox\fitColB=\vsplit\fitTmp to #2
    \dp\fitColA=0pt \dp\fitColB=0pt
    \setbox0=\hbox{%
      \vbox{\nointerlineskip \prevdepth=-1000pt \box\fitColA}%
      \hskip\columnsep
      \vbox{\nointerlineskip \prevdepth=-1000pt \box\fitColB}%
    }%
    \ifvoid\fitTmp
      \fitMeasureDim=#2\relax
    \else
      \fitMeasureDim=\dimexpr #2+1pt\relax
    \fi
  }%
  % --- coarse shrink until the height fits ---
  \def\fitShrink{%
    \loop
      \fitFontDim.5\fitFontDim \fitBuild
    \ifdim\fitMeasureDim>#2
      \ifdim\fitFontDim>.1pt \repeat
    \fi
    \fitLower\fitFontDim \fitUpper2\fitFontDim \fitFontDim1.5\fitFontDim
  }%
  % --- fine tuning ---
  \fitBuild
  \ifdim\fitMeasureDim>#2 \fitShrink\fi
  \loop
    \fitBuild
    \ifdim\fitMeasureDim>#2
      \ifdim\fitFontDim>.2pt
        \fitMoretrue
        \fitUpper\fitFontDim \advance\fitFontDim\fitLower \fitFontDim.5\fitFontDim
      \fi
    \else
      \advance\fitMeasureDim-#2
      \ifdim\fitMeasureDim<10pt
        \fitLower\fitFontDim \advance\fitFontDim\fitUpper \fitFontDim.5\fitFontDim
        \fitMeasureDim=\fitUpper \advance\fitMeasureDim-\fitLower
        \ifdim\fitMeasureDim<.2pt \fitMorefalse \else \fitMoretrue \fi
      \else
        \fitMorefalse
      \fi
    \fi
  \iffitMore \repeat
}
{%
  % final output: fixed height, blank space at the bottom
  \dp0=0pt
  \vbox to #2{%
    \nointerlineskip \prevdepth=-1000pt
    \box0
    \vfil
  }%
}

	\fontsize{\fitFontDim}{1.2\fitFontDim}\selectfont
	\fitStuff
	}%

      \hsize=#1\relax
      \parskip=0pt
      \fontsize{\fitFontDim}{1.2\fitFontDim}\selectfont
      \fitStuff\par

      \fitMeasureDim=\ht0
	\ifdim\fitLineWidth>#1
	\fitMeasureDim=\dimexpr #2+1pt\relax % erzwingt Shrink-Runde
	\fi  
  }%
